cmake_minimum_required(VERSION 3.16)
project(resting_order_book LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Explicitly override the usual Release -O3, also record compile_commands.json.
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -DNDEBUG")
set(CMAKE_CXX_FLAGS_DEBUG "-O0 -g")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
add_executable(bench src/main.cpp src/verify.cpp)
target_include_directories(bench PRIVATE include src)
target_compile_options(bench PRIVATE -Wall -Wextra -Wpedantic -g -fno-omit-frame-pointer)
if(SANITIZE)
  target_compile_options(bench PRIVATE -fsanitize=address,undefined)
  target_link_options(bench PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME bounded_correctness COMMAND bench --verify --ops 10000)
